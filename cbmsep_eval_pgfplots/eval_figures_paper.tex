  \addplot[fill=cbblue, draw=cbblue] coordinates

  \node[font=\scriptsize, anchor=north west] at (rel axis cs:0.03,0.97)
    {on-chain overhead $+21\%$};
  \end{axis}
  \end{tikzpicture}
  \caption{Cost breakdown. In-chaincode verification adds $21\%$ over a native

  per attestation.}
  \label{fig:overhead}
\end{figure}
